void createContact(AddressBook *addressBook)
{
char name[50];
char phone[20];
char email[50];   
int i;
int valid;
int at_count;
int dot_count;
int at_pos;
int dot_pos;
int attempt=0;
int len;
 /* ---------------- NAME VALIDATION ---------------- */
while(attempt < 3)
{
    if(attempt == 0)
        printf("Enter the name: ");
    else
        printf("Re-enter the name: ");
    scanf(" %49[^\n]", name);
    len = strlen(name);
    if(len < 4)
    {
        printf("Error Message : Name must contain at least 4 characters\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    valid = 1;
    for(i = 0; name[i] != '\0'; i++)
    {
        if(!isalpha(name[i]) && name[i] != ' ')
        {
            valid = 0;
            break;
        }
    }
    if(valid == 0)
    {
        printf("Error Message : Name should contain only alphabets\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    break;
}
if(attempt == 3)
{
    printf("Maximum attempts reached. Exiting Create Contact.\n");
    return;
}
/* ---------------- PHONE VALIDATION ---------------- */
attempt = 0;
while(attempt < 3)
{
    if(attempt == 0)
        printf("Enter the phone number: ");
    else
        printf("Re-enter the phone number: ");
    scanf("%19s", phone);
    len = strlen(phone);
    /* Check exactly 10 digits */
    if(len != 10)
    {
        printf("Error Message : Phone number must contain exactly 10 digits\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* First digit must be 6 to 9 */
    if(phone[0] < '6' || phone[0] > '9')
    {
        printf("Error Message : First digit must be between 6 and 9\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Only digits allowed */
    valid = 1;
    for(i = 0; phone[i] != '\0'; i++)
    {
        if(!isdigit(phone[i]))
        {
            valid = 0;
            break;
        }
    }
    if(valid == 0)
    {
        printf("Error Message : Other than digits any other character not allowed in phone number\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Check phone number uniqueness */
    valid = 1;
    for(i = 0; i < addressBook->contactCount; i++)
    {
        if(strcmp(addressBook->contacts[i].phone, phone) == 0)
        {
            valid = 0;
            break;
        }
    }
    if(valid == 0)
    {
        printf("Error Message : Phone number already exists\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    break;
}
if(attempt == 3)
{
    printf("Maximum attempts reached. Exiting Create Contact.\n");
    return;
}
/* ---------------- EMAIL VALIDATION ---------------- */
attempt = 0;
while(attempt < 3)
{
    if(attempt == 0)
        printf("Enter the email: ");
    else
        printf("Re-enter the email: ");
    scanf("%49s", email);
    at_count = 0;
    dot_count = 0;
    at_pos = -1;
    dot_pos = -1;
    valid = 1;
    len = strlen(email);
    for(i = 0; email[i] != '\0'; i++)
    {
        /* Only lowercase letters, digits, @ and . allowed */
        if(!islower(email[i]) && !isdigit(email[i]) &&
           email[i] != '@' && email[i] != '.')
        {
            printf("Error Message : Invalid symbol\n");
            valid = 0;
            break;
        }
        if(email[i] == '@')
        {
            at_count++;
            at_pos = i;
        }
        if(email[i] == '.')
        {
            dot_count++;
            dot_pos = i;
        }
    }
    if(valid == 0)
    {
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Exactly one @ */
    if(at_count == 0)
    {
        printf("Error Message : Missing @\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    if(at_count > 1)
    {
        printf("Error Message : Multiple @ symbols are not allowed\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Exactly one dot */
    if(dot_count == 0)
    {
        printf("Error Message : Missing . (dot)\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    if(dot_count > 1)
    {
        printf("Error Message : Email must contain exactly one . (dot)\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Dot must come after @ */
    if(dot_pos < at_pos)
    {
        printf("Error Message : Dot must appear after @\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* At least one character between @ and . */
    if(dot_pos == at_pos + 1)
    {
        printf("Error Message : At least one character must be present between @ and .\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Something must exist before @ */
    if(at_pos == 0)
    {
        printf("Error Message : Invalid email ID\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* .com must be at the end */
    if(strcmp(&email[dot_pos], ".com") != 0)
    {
        printf("Error Message : Email must end with .com\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    /* Check email uniqueness */
    valid = 1;
    for(i = 0; i < addressBook->contactCount; i++)
    {
        if(strcmp(addressBook->contacts[i].email, email) == 0)
        {
            valid = 0;
            break;
        }
    }
    if(valid == 0)
    {
        printf("Error Message : Email ID already exists\n");
        attempt++;
        printf("Attempts left: %d\n", 3 - attempt);
        continue;
    }
    break;
}
if(attempt == 3)
{
    printf("Maximum attempts reached. Exiting Create Contact.\n");
    return;
}
strcpy(addressBook->contacts[addressBook->contactCount].name, name);
strcpy(addressBook->contacts[addressBook->contactCount].phone, phone);
strcpy(addressBook->contacts[addressBook->contactCount].email, email);
addressBook->contactCount++;
printf("Contact added successfully!\n");
}

void editContact(AddressBook *addressBook)
{

    int editChoice;
    int i;
    int j;
    int found;
    int matchCount;
    int selectedIndex = -1;
    int choice;
    char searchName[50];
    char searchPhone[20];
    char searchEmail[50];

    char phone[20];
    char email[50];

                       SEARCH MENU

    printf("2. Search by Phone Number\n");
    printf("3. Search by Email ID\n");
    printf("4. Exit\n");

    {
        printf("Error Message : Invalid choice. Please enter 1 to 4.\n");

        return;
    }

        return;

                       SEARCH BY PHONE

        {

        }
        for(i = 0; searchPhone[i] != '\0'; i++)
        {
            if(!isdigit(searchPhone[i]))
            {
                printf("Error Message : Phone number should contain only digits.\n");
                return;
            }
        }
        /* Find matching contacts */
        for(i = 0; i < addressBook->contactCount; i++)
        {
            if(strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
            {
                matchCount++;
            }
        }
    }
    /* =====================================================
                       SEARCH BY EMAIL
       ===================================================== */
    else if(searchChoice == 3)
    {
        int at_count = 0;
        int dot_count = 0;
        int at_pos = -1;
        int dot_pos = -1;
        printf("Enter the email: ");
        scanf("%49s", searchEmail);
        for(i = 0; searchEmail[i] != '\0'; i++)
        {
            if(!islower(searchEmail[i]) &&
               !isdigit(searchEmail[i]) &&
               searchEmail[i] != '@' &&
               searchEmail[i] != '.')
            {
                printf("Error Message : Invalid symbol.\n");
                return;
            }
            if(searchEmail[i] == '@')
            {
                at_count++;
                at_pos = i;
            }
            if(searchEmail[i] == '.')
            {
                dot_count++;
                dot_pos = i;
            }
        }
        if(at_count != 1)
        {
            printf("Error Message : Email must contain exactly one @.\n");
            return;
        }
        if(dot_count != 1)
        {
            printf("Error Message : Email must contain exactly one dot.\n");
            return;
        }
        if(at_pos == 0)
        {
            printf("Error Message : Invalid email ID.\n");
            return;
        }
        if(dot_pos <= at_pos + 1)
        {
            printf("Error Message : Invalid email format.\n");
            return;
        }
        if(strcmp(&searchEmail[dot_pos], ".com") != 0)
        {
            printf("Error Message : Email must end with .com.\n");
            return;
        }
        /* Find matching contacts */
        for(i = 0; i < addressBook->contactCount; i++)
        {
            if(strcmp(addressBook->contacts[i].email, searchEmail) == 0)
            {
                matchCount++;
            }
        }
    }
    /* =====================================================
                       NO MATCH FOUND
       ===================================================== */
    if(matchCount == 0)
    {
        printf("No matching contact found.\n");
        return;
    }
    /* =====================================================
                  DISPLAY MATCHING CONTACTS
       ===================================================== */
    printf("\n--------- MATCHING CONTACTS ---------\n");
    j = 1;
    for(i = 0; i < addressBook->contactCount; i++)
    {
        found = 0;
        if(searchChoice == 1 &&
           strcmp(addressBook->contacts[i].name, searchName) == 0)
        {
            found = 1;
        }
        else if(searchChoice == 2 &&
                strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
        {
            found = 1;
        }
        else if(searchChoice == 3 &&
                strcmp(addressBook->contacts[i].email, searchEmail) == 0)
        {
            found = 1;
        }
        if(found)
        {
            printf("%d. Name: %s | Phone: %s | Email: %s\n",
                   j,
                   addressBook->contacts[i].name,
                   addressBook->contacts[i].phone,
                   addressBook->contacts[i].email);
            j++;
        }
    }
    /* =====================================================
                  SELECT CONTACT TO EDIT
       ===================================================== */
    if(matchCount == 1)
    {
        /* Only one matching record */
        for(i = 0; i < addressBook->contactCount; i++)
        {
            if((searchChoice == 1 &&
                strcmp(addressBook->contacts[i].name, searchName) == 0) ||
               (searchChoice == 2 &&
                strcmp(addressBook->contacts[i].phone, searchPhone) == 0) ||
               (searchChoice == 3 &&
                strcmp(addressBook->contacts[i].email, searchEmail) == 0))
            {
                selectedIndex = i;
                break;
            }
        }
    }
    else
    {
        /* Multiple records found */
        printf("\nMultiple matching records found.\n");
        printf("Select the record to edit: ");
        /* Prevent infinite loop */
        if(scanf("%d", &choice) != 1)
        {
            printf("Error Message : Invalid record selection.\n");
            while(getchar() != '\n');
            return;
        }
        if(choice < 1 || choice > matchCount)

            printf("Error Message : Invalid record selection.\n");
            return;
        }
        j = 1;
        for(i = 0; i < addressBook->contactCount; i++)

            found = 0;
            if(searchChoice == 1 &&
               strcmp(addressBook->contacts[i].name, searchName) == 0)
            {
                found = 1;
            }
            else if(searchChoice == 2 &&
                    strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
            {
                found = 1;
            }
            else if(searchChoice == 3 &&
                    strcmp(addressBook->contacts[i].email, searchEmail) == 0)
            {
                found = 1;
            }
            if(found)
            {
                if(j == choice)
                {
                    selectedIndex = i;
                    break;
                }
                j++;
            }
        }
    }
    /* Safety check */
    if(selectedIndex == -1)
    {
        printf("Error Message : Contact selection failed.\n");
        return;
    }
    /* =====================================================
                       EDIT MENU
       ===================================================== */
    while(1)
    {
        printf("\n-------- EDIT CONTACT --------\n");
        printf("Current Name  : %s\n",
               addressBook->contacts[selectedIndex].name);
        printf("Current Phone : %s\n",
               addressBook->contacts[selectedIndex].phone);
        printf("Current Email : %s\n",
               addressBook->contacts[selectedIndex].email);

        printf("2. Edit Phone Number\n");
        printf("3. Edit Email ID\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        /*
         * IMPORTANT:
         * scanf returns 1 when an integer is successfully read.
         * If user enters abc, scanf returns 0.
         */
        if(scanf("%d", &editChoice) != 1)
        {
            printf("Error Message : Invalid choice. Please enter 1 to 4.\n");
            /* Clear invalid input from buffer */
            while(getchar() != '\n');
            continue;
        }
        /* Check range */
        if(editChoice < 1 || editChoice > 4)
        {
            printf("Error Message : Invalid choice. Please enter 1 to 4.\n");
            continue;
        }
        /* Exit */
        if(editChoice == 4)
        {
            printf("Exiting edit menu...\n");
            return;
        }
        /* =================================================
                         EDIT NAME
           ================================================= */
        if(editChoice == 1)
        {
            printf("Enter new name: ");
            scanf(" %49[^\n]", name);
            /* Name length validation */
            if(strlen(name) < 4)
            {
                printf("Error Message : Name must contain at least 4 characters.\n");
                continue;
            }
            /* Name character validation */
            for(i = 0; name[i] != '\0'; i++)
            {
                if(!isalpha(name[i]) && name[i] != ' ')
                {
                    printf("Error Message : Name should contain only alphabets.\n");
                    break;
                }
            }
            if(name[i] != '\0')
            {
                continue;
            }
            /* Update name */
            strcpy(addressBook->contacts[selectedIndex].name, name);
            printf("Name updated successfully.\n");
        }
        /* =================================================
                         EDIT PHONE
           ================================================= */
        else if(editChoice == 2)
        {
            printf("Enter new phone number: ");
            scanf("%19s", phone);
            /* Length validation */
            if(strlen(phone) != 10)
            {
                printf("Error Message : Phone number must contain exactly 10 digits.\n");
                continue;
            }
            /* First digit validation */
            if(phone[0] < '6' || phone[0] > '9')
            {
                printf("Error Message : First digit must be between 6 and 9.\n");
                continue;
            }
            /* Digit validation */
            for(i = 0; phone[i] != '\0'; i++)
            {
                if(!isdigit(phone[i]))
                {
                    printf("Error Message : Phone number should contain only digits.\n");
                    break;
                }
            }
            if(phone[i] != '\0')
            {
                continue;
            }
            /* Duplicate phone check */
            found = 0;
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if(i != selectedIndex &&
                   strcmp(addressBook->contacts[i].phone, phone) == 0)
                {
                    found = 1;
                    break;
                }
            }
            if(found)
            {
                printf("Error Message : Phone number already exists.\n");
                continue;
            }
            /* Update phone */
            strcpy(addressBook->contacts[selectedIndex].phone, phone);
            printf("Phone number updated successfully.\n");
        }
        /* =================================================
                         EDIT EMAIL
           ================================================= */
        else if(editChoice == 3)
        {
            int at_count = 0;
            int dot_count = 0;
            int at_pos = -1;
            int dot_pos = -1;
            printf("Enter new email: ");
            scanf("%49s", email);
            /* Email character validation */
            for(i = 0; email[i] != '\0'; i++)
            {
                if(!islower(email[i]) &&
                   !isdigit(email[i]) &&
                   email[i] != '@' &&
                   email[i] != '.')
                {
                    printf("Error Message : Invalid symbol.\n");
                    break;
                }
                if(email[i] == '@')
                {
                    at_count++;
                    at_pos = i;
                }
                if(email[i] == '.')
                {
                    dot_count++;
                    dot_pos = i;
                }
            }
            if(email[i] != '\0')
            {
                continue;
            }
            /* Exactly one @ */
            if(at_count != 1)
            {
                printf("Error Message : Email must contain exactly one @.\n");
                continue;
            }
            /* Exactly one dot */
            if(dot_count != 1)
            {
                printf("Error Message : Email must contain exactly one dot.\n");
                continue;
            }
            /* Something before @ */
            if(at_pos == 0)
            {
                printf("Error Message : Invalid email ID.\n");
                continue;
            }
            /* Character between @ and . */
            if(dot_pos <= at_pos + 1)
            {
                printf("Error Message : At least one character must be present between @ and .\n");
                continue;
            }
            /* .com validation */
            if(strcmp(&email[dot_pos], ".com") != 0)
            {
                printf("Error Message : Email must end with .com.\n");
                continue;
            }
            /* Duplicate email check */
            found = 0;
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if(i != selectedIndex &&
                   strcmp(addressBook->contacts[i].email, email) == 0)
                {
                    found = 1;
                    break;
                }
            }
            if(found)
            {
                printf("Error Message : Email ID already exists.\n");
                continue;
            }
            /* Update email */
            strcpy(addressBook->contacts[selectedIndex].email, email);
            printf("Email updated successfully.\n");
        }
    }  
}
void deleteContact(AddressBook *addressBook)
{
    int searchChoice;
    int i, j;
    int found;
    int matchCount;
    int selectedIndex;
    int choice;
    char confirm;
    char searchName[50];
    char searchPhone[20];
    char searchEmail[50];
    /* ---------------- DELETE MENU ---------------- */
    while(1)
    {
        printf("\n-------- DELETE CONTACT --------\n");
        printf("1. Delete by Name\n");
        printf("2. Delete by Phone Number\n");
        printf("3. Delete by Email ID\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &searchChoice);
        /* Validate menu choice */
        if(searchChoice < 1 || searchChoice > 4)
        {
            printf("Error Message : Invalid choice. Please enter 1 to 4.\n");
            continue;
        }
        /* Exit */
        if(searchChoice == 4)
        {
            printf("Exiting delete menu...\n");
            return;
        }
        matchCount = 0;
        /* ================================================= */
        /* DELETE BY NAME                                    */
        /* ================================================= */
        if(searchChoice == 1)
        {
            printf("Enter the name: ");
            scanf(" %49[^\n]", searchName);
            /* Name validation */
            if(strlen(searchName) < 4)
            {
                printf("Error Message : Name must contain at least 4 characters.\n");
                continue;
            }
            for(i = 0; searchName[i] != '\0'; i++)
            {
                if(!isalpha(searchName[i]) && searchName[i] != ' ')
                {
                    printf("Error Message : Name should contain only alphabets.\n");
                    break;
                }
            }
            if(searchName[i] != '\0')
            {
                continue;
            }
            /* Find matching records */
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if(strcmp(addressBook->contacts[i].name, searchName) == 0)
                {
                    matchCount++;
                }
            }
        }
        /* ================================================= */
        /* DELETE BY PHONE                                   */
        /* ================================================= */
        else if(searchChoice == 2)
        {
            printf("Enter the phone number: ");
            scanf("%19s", searchPhone);
            /* Phone validation */
            if(strlen(searchPhone) != 10)
            {
                printf("Error Message : Phone number must contain exactly 10 digits.\n");
                continue;
            }
            if(searchPhone[0] < '6' || searchPhone[0] > '9')
            {
                printf("Error Message : First digit must be between 6 and 9.\n");
                continue;
            }
            for(i = 0; searchPhone[i] != '\0'; i++)
            {
                if(!isdigit(searchPhone[i]))
                {
                    printf("Error Message : Phone number should contain only digits.\n");
                    break;
                }
            }
            if(searchPhone[i] != '\0')
            {
                continue;
            }
            /* Find matching records */
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if(strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
                {
                    matchCount++;
                }
            }
        }
        /* ================================================= */
        /* DELETE BY EMAIL                                   */
        /* ================================================= */
        else if(searchChoice == 3)
        {
            int at_count = 0;
            int dot_count = 0;
            int at_pos = -1;
            int dot_pos = -1;
            printf("Enter the email: ");
            scanf("%49s", searchEmail);
            /* Email validation */
            for(i = 0; searchEmail[i] != '\0'; i++)
            {
                if(!islower(searchEmail[i]) &&
                   !isdigit(searchEmail[i]) &&
                   searchEmail[i] != '@' &&
                   searchEmail[i] != '.')
                {
                    printf("Error Message : Invalid symbol.\n");
                    break;
                }
                if(searchEmail[i] == '@')
                {
                    at_count++;
                    at_pos = i;
                }
                if(searchEmail[i] == '.')
                {
                    dot_count++;
                    dot_pos = i;
                }
            }
            if(searchEmail[i] != '\0')
            {
                continue;
            }
            if(at_count != 1)
            {
                printf("Error Message : Email must contain exactly one @.\n");
                continue;
            }
            if(dot_count != 1)
            {
                printf("Error Message : Email must contain exactly one dot.\n");
                continue;
            }
            if(at_pos == 0)
            {
                printf("Error Message : Invalid email ID.\n");
                continue;
            }
            if(dot_pos <= at_pos + 1)
            {
                printf("Error Message : Invalid email format.\n");
                continue;
            }
            if(strcmp(&searchEmail[dot_pos], ".com") != 0)
            {
                printf("Error Message : Email must end with .com.\n");
                continue;
            }
            /* Find matching records */
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if(strcmp(addressBook->contacts[i].email, searchEmail) == 0)
                {
                    matchCount++;
                }
            }
        }
        /* ================================================= */
        /* NO MATCH                                          */
        /* ================================================= */
        if(matchCount == 0)
        {
            printf("No matching contact found.\n");
            continue;
        }
        /* ================================================= */
        /* DISPLAY MATCHING RECORDS                          */
        /* ================================================= */
        printf("\n--------- MATCHING CONTACTS ---------\n");
        j = 1;
        for(i = 0; i < addressBook->contactCount; i++)
        {
            found = 0;
            if(searchChoice == 1 &&
               strcmp(addressBook->contacts[i].name, searchName) == 0)
            {
                found = 1;
            }
            else if(searchChoice == 2 &&
                    strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
            {
                found = 1;
            }
            else if(searchChoice == 3 &&
                    strcmp(addressBook->contacts[i].email, searchEmail) == 0)
            {
                found = 1;
            }
            if(found)
            {
                printf("%d. Name: %s | Phone: %s | Email: %s\n",
                       j,
                       addressBook->contacts[i].name,
                       addressBook->contacts[i].phone,
                       addressBook->contacts[i].email);
                j++;
            }
        }
        /* ================================================= */
        /* SELECT RECORD                                    */
        /* ================================================= */
        if(matchCount == 1)
        {
            /* Only one record found */
            for(i = 0; i < addressBook->contactCount; i++)
            {
                if((searchChoice == 1 &&
                    strcmp(addressBook->contacts[i].name, searchName) == 0) ||
                   (searchChoice == 2 &&
                    strcmp(addressBook->contacts[i].phone, searchPhone) == 0) ||
                   (searchChoice == 3 &&
                    strcmp(addressBook->contacts[i].email, searchEmail) == 0))
                {
                    selectedIndex = i;
                    break;
                }
            }
        }
        else
        {
            /* Multiple records found */
            printf("\nMultiple matching records found.\n");
            printf("Select the record to delete: ");
            scanf("%d", &choice);
            if(choice < 1 || choice > matchCount)
            {
                printf("Error Message : Invalid record selection.\n");
                continue;
            }
            j = 1;
            for(i = 0; i < addressBook->contactCount; i++)
            {
                found = 0;
                if(searchChoice == 1 &&
                   strcmp(addressBook->contacts[i].name, searchName) == 0)
                {
                    found = 1;
                }
                else if(searchChoice == 2 &&
                        strcmp(addressBook->contacts[i].phone, searchPhone) == 0)
                {
                    found = 1;
                }
                else if(searchChoice == 3 &&
                        strcmp(addressBook->contacts[i].email, searchEmail) == 0)
                {
                    found = 1;
                }
                if(found)
                {
                    if(j == choice)
                    {
                        selectedIndex = i;
                        break;
                    }
                    j++;
                }
            }
        }
        /* ================================================= */
        /* CONFIRMATION BEFORE DELETE                        */
        /* ================================================= */
        printf("\nContact selected for deletion:\n");
        printf("Name  : %s\n",
               addressBook->contacts[selectedIndex].name);
        printf("Phone : %s\n",
               addressBook->contacts[selectedIndex].phone);
        printf("Email : %s\n",
               addressBook->contacts[selectedIndex].email);
        printf("\nAre you sure you want to delete this contact? (y/n): ");
        scanf(" %c", &confirm);
        if(confirm != 'y' && confirm != 'Y' &&
           confirm != 'n' && confirm != 'N')
        {
            printf("Error Message : Invalid confirmation choice.\n");
            continue;
        }
        /* User cancelled deletion */
        if(confirm == 'n' || confirm == 'N')
        {
            printf("Deletion cancelled.\n");
            continue;
        }
        /* ================================================= */
        /* DELETE CONTACT                                    */
        /* ================================================= */
        for(i = selectedIndex; i < addressBook->contactCount - 1; i++)
        {
            addressBook->contacts[i] = addressBook->contacts[i + 1];
        }
        addressBook->contactCount--;
        printf("Contact deleted successfully.\n");
    }
